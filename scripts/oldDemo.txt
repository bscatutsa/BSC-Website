//membership tier
  function getMembershipTier(points) {
    if (points >= 36) return { tier: "Executive", class: "tier-executive" };
    if (points >= 21) return { tier: "Professional", class: "tier-professional" };
    if (points >= 11) return { tier: "Intern", class: "tier-intern" };
    return { tier: "Member", class: "tier-member" };
  }

// --------------------------------------------------
// DONUT CHART RENDER
// --------------------------------------------------
function renderChart(member) {
  const ctx = document.getElementById("pointsChart").getContext("2d");

  const cats = member.categories;

  const labels = [];
  const data = [];
  const colors = [];

  const colorMap = {
    socials: "#5DA8FF",
    meetings: "#1D5DFF",
    reposts: "#7BC8FF",
    fundraiser: "#FF914D",
    volunteering: "#FFCE56",
    workshops: "#FFC0CB",
    events: "#9B59B6",
    extra: "#2ECC71",
    committee: "#34495E"
  };

  for (const key in cats) {
    const val = cats[key];
    if (val > 0) {
      labels.push(key.charAt(0).toUpperCase() + key.slice(1));
      data.push(val);
      colors.push(colorMap[key]);
    }
  }

  if (pointsChart) pointsChart.destroy();

  pointsChart = new Chart(ctx, {
    type: "doughnut",
    data: {
      labels,
      datasets: [
        {
          data,
          backgroundColor: colors,
          hoverOffset: 12
        }
      ]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      animation: {
        animateRotate: true,
        animateScale: true,
        duration: 900
      },
      plugins: {
        tooltip: {
          callbacks: {
            label: function (context) {
              const val = context.raw;
              const total = data.reduce((a, b) => a + b, 0);
              const pct = ((val / total) * 100).toFixed(1);
              return `${context.label}: ${val} (${pct}%)`;
            }
          }
        },
        legend: {
          position: "bottom",
          labels: { boxWidth: 15 }
        }
      }
    }
  });

  // Summary Text
  const sorted = Object.entries(cats)
    .filter(([k, v]) => v > 0)
    .sort((a, b) => b[1] - a[1]);

  let summaryText = "";

  if (sorted.length === 0) {
    summaryText = "No points earned yet.";
  } else {
    const [topCat, topVal] = sorted[0];
    summaryText += `Most points from ${topCat} (${topVal}). `;

    if (sorted[1]) {
      const [secCat, secVal] = sorted[1];
      summaryText += `Second highest from ${secCat} (${secVal}). `;
    }

    const zeroCats = Object.entries(cats)
      .filter(([k, v]) => v === 0)
      .map(([k]) => k);

    if (zeroCats.length > 0) {
      summaryText += `No points in: ${zeroCats.join(", ")}.`;
    }
  }

}

// BAR CHART RENDER
// --------------------------------------------------
function renderBarChart(member) {
  const ctx = document.getElementById("pointsChart").getContext("2d");

  const cats = member.categories;
  const labels = [];
  const data = [];

  for (const key in cats) {
    if (cats[key] > 0) {
      labels.push(key.charAt(0).toUpperCase() + key.slice(1));
      data.push(cats[key]);
    }
  }

  if (pointsChart) pointsChart.destroy();

  pointsChart = new Chart(ctx, {
    type: "bar",
    data: {
      labels,
      datasets: [
        {
          data,
          backgroundColor: labels.map(label => {
            const key = label.toLowerCase();
            return {
              socials:      "#5DA8FF",
              meetings:     "#1D5DFF",
              reposts:      "#7BC8FF",
              fundraiser:   "#FF914D",
              volunteering: "#FFCE56",
              workshops:    "#FFC0CB",
              events:       "#9B59B6",
              extra:        "#2ECC71",
              committee:    "#34495E"
            }[key] || "#1A4B89";   // fallback
          }),
          borderRadius: 6
        }
      ]
    },
    options: {
      indexAxis: "y",
      scales: {
        x: { beginAtZero: true }
      },
      plugins: {
        legend: { display: false }
      },
      responsive: true,
      maintainAspectRatio: false
    }
  });
}

function renderChartByMode(member) {
  if (chartMode === "donut") {
    renderChart(member);
  } else {
    renderBarChart(member);
  }
}

// LOAD + PARSE SHEET
// --------------------------------------------------
async function loadMembersFromSheet() {
  try {
    const res = await fetch(SHEET_URL);
    const csv = await res.text();

    const rows = csv.split(/\r?\n/).map(r => r.split(","));
    const header = rows[0].map(h => h.trim());

    let db = {};

    for (let i = 1; i < rows.length; i++) {
      const row = rows[i];
      if (!row[0] || !row[2]) continue;

      const first = row[0].trim();
      const last = row[1].trim();
      const id = row[2].trim().toLowerCase();
      const total = parseInt(row[3] || 0);

      const socials = parseInt(row[4] || 0);
      const meetings = parseInt(row[5] || 0);
      const reposts = parseInt(row[6] || 0);
      const fundraiser = parseInt(row[7] || 0);
      const volunteering = parseInt(row[8] || 0);
      const workshops = parseInt(row[9] || 0);
      const events = parseInt(row[10] || 0);
      const extra = parseInt(row[11] || 0);
      const committee = parseInt(row[12] || 0);

      db[id] = {
        utsaId: id,
        name: `${first} ${last}`,
        first,
        last,
        points: total,
        originalPoints: total,
        categories: {
          socials,
          meetings,
          reposts,
          fundraiser,
          volunteering,
          workshops,
          events,
          extra,
          committee
        },
        status: "Active",
        eventsList: []
      };
    }

    memberDB = db;
  } catch (err) {
    console.error("CSV PARSE FAILED:", err);
  }
}

// HELPER FUNCTIONS

function setStatusPill(statusText) {
  const pill = document.getElementById("memberStatus");
  pill.textContent = statusText || "—";
  pill.classList.remove("active", "probation");

  const s = (statusText || "").toLowerCase();
  if (s.includes("active")) pill.classList.add("active");
  if (s.includes("probation")) pill.classList.add("probation");
}

function showMessage(id, text, type) {
  const el = document.getElementById(id);
  if (!el) return;
  el.textContent = text;
  el.classList.remove("ok", "error");
  if (type === "ok") el.classList.add("ok");
  if (type === "error") el.classList.add("error");
}

function animatePoints(from, to) {
  const el = document.getElementById("memberPoints");
  const duration = 500;
  const start = performance.now();

  function step(ts) {
    const progress = Math.min((ts - start) / duration, 1);
    el.textContent = Math.round(from + (to - from) * progress);
    if (progress < 1) requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
}

function renderEvents(member) {
  const eventList = document.getElementById("eventList");
  const filterRow = document.getElementById("eventFilters");

  // Safety check
  if (!eventList || !filterRow) return;

  eventList.innerHTML = "";
  filterRow.innerHTML = "";

  const events = member.events || [];

  // No events?
  if (events.length === 0) {
      eventList.textContent = "No events recorded.";
      return;
  }

  // Generate list
  events.forEach(ev => {
      const row = document.createElement("div");
      row.textContent = ev;
      row.style.marginBottom = "4px";
      eventList.appendChild(row);
  });

  // Simple ALL filter so the UI doesn't break
  const allPill = document.createElement("div");
  allPill.className = "event-filter-pill active";
  allPill.textContent = "All";
  filterRow.appendChild(allPill);
}


function renderCategoryBadges(member) {
  const container = document.getElementById("categoryBadges");
  if (!container) return;

  container.innerHTML = "";

  // These MUST match the chart dataset order + colors
  const categories = [
      { key: "socials", label: "Socials", color: "#4F8AFF" },
      { key: "meetings", label: "Meetings", color: "#7C4DFF" },
      { key: "workshops", label: "Workshops", color: "#FFB2C4" },
      { key: "events", label: "Events", color: "#54C1A9" },
      { key: "extra", label: "Extra", color: "#7BD13A" }
  ];

  categories.forEach(cat => {
      const points = member.pointsByCategory?.[cat.key] || 0;

      // Show only categories that actually have points
      if (points > 0) {
          const badge = document.createElement("span");
          badge.className = "category-badge";
          badge.textContent = `${cat.label}: ${points}`;
          badge.style.backgroundColor = cat.color;
          container.appendChild(badge);
      }
  });
}

// --------------------------------------------------
// RENDER MEMBER CARD
// --------------------------------------------------
function renderMember(member) {
  const card = document.getElementById("memberCard");

  // 1) If no member loaded → hide card
  if (!member) {
    card.style.display = "none";
    return;
  }

  // Show card
  card.style.display = "block";

  // 2) BASIC INFO
  document.getElementById("memberName").textContent = member.name;
  document.getElementById("memberPoints").textContent = member.points;
  setStatusPill(member.status || "");

  // 3) MEMBERSHIP TIER BADGE
  const tierInfo = getMembershipTier(member.points);
  const badge = document.getElementById("membershipTierBadge");
  badge.textContent = tierInfo.tier;
  badge.className = tierInfo.class;

  // 4) RENDER CHART (Donut or Bar)
  renderChartByMode(member);

  // 5) RENDER CATEGORY BADGES (under chart)
  renderCategoryBadges(member);

  // 6) RENDER EVENTS LIST + CATEGORY FILTER BUTTONS
  renderEvents(member);
}

// DOM LOGIC

document.addEventListener("DOMContentLoaded", () => {
  const utsaInput = document.getElementById("utsaIdInput");
  const lookupBtn = document.getElementById("lookupBtn");
  const checkInBtn = document.getElementById("checkInBtn");
  const adminToggle = document.getElementById("adminToggle");
  const adminControls = document.getElementById("adminControls");
  const attendanceList = document.getElementById("attendanceList");
  const attendanceEmpty = document.getElementById("attendanceEmpty");
  const resetPointsBtn = document.getElementById("resetPointsBtn");

  loadMembersFromSheet();

  // CHART TOGGLE
  const toggleTrack = document.getElementById("chartToggleTrack");
  const toggleKnob = document.getElementById("chartToggleKnob");
  const toggleOptions = document.querySelectorAll(".chart-toggle-option");

  toggleOptions.forEach(opt => {
    opt.addEventListener("click", () => {
      chartMode = opt.dataset.mode;

      // move knob
      if (chartMode === "donut") {
        toggleKnob.style.left = "3px";
      } else {
        toggleKnob.style.left = "107px";
      }

      toggleOptions.forEach(o => o.classList.remove("active"));
      opt.classList.add("active");

      if (currentMember) {
        renderChartByMode(currentMember);
      }
    });
  });

  // LOOKUP
  function doLookup() {
    const id = (utsaInput.value || "").trim().toLowerCase();

    if (!id) {
      showMessage("lookupMessage", "Enter a myUTSA ID.", "error");
      renderMember(null);
      currentMember = null;
      return;
    }

    if (!memberDB[id]) {
      showMessage("lookupMessage", `No member found for "${id}".`, "error");
      renderMember(null);
      currentMember = null;
      return;
    }

    currentMember = { ...memberDB[id] };
    showMessage("lookupMessage", `Loaded ${currentMember.name}.`, "ok");
    showMessage("checkInMessage", "", "ok");
    renderMember(currentMember);
  }

  lookupBtn.addEventListener("click", doLookup);
  utsaInput.addEventListener("keydown", e => {
    if (e.key === "Enter") doLookup();
  });

  document.querySelectorAll(".hidden").forEach(el => {
    el.classList.remove("hidden");
  });

  // CHECK-IN
  checkInBtn.addEventListener("click", () => {
    if (!currentMember) {
      showMessage("checkInMessage", "Lookup a member first.", "error");
      return;
    }

    if (attendanceSet.has(currentMember.utsaId)) {
      showMessage("checkInMessage", "Already checked in.", "error");
      return;
    }

    attendanceSet.add(currentMember.utsaId);
    showMessage("checkInMessage", "Check-in recorded.", "ok");

    if (attendanceEmpty) attendanceEmpty.remove();

    const li = document.createElement("li");
    li.textContent = `${currentMember.name} (${currentMember.utsaId})`;
    attendanceList.prepend(li);
  });

  // ADMIN MODE
  adminToggle.addEventListener("change", () => {
    adminControls.style.display = adminToggle.checked ? "block" : "none";
  });

  adminControls.addEventListener("click", e => {
    const btn = e.target.closest("button");
    if (!btn || !currentMember) return;

    if (btn === resetPointsBtn) {
      animatePoints(currentMember.points, currentMember.originalPoints);
      currentMember.points = currentMember.originalPoints;
      return;
    }

    const delta = parseInt(btn.getAttribute("data-delta")) || 0;
    const newVal = currentMember.points + delta;
    animatePoints(currentMember.points, newVal);
    currentMember.points = newVal;
  });
});
